private function assignment_() 
	dim expr_ as new Expr
	st expr_, or_f()
	
	if match(EQUAL) then
		dim eqs as new Token
		st eqs, previous()
		
		dim value as new Expr
		st value, assignment_()
		if (typename(expr_) = "VariableExpr") then
			dim varexprcast as VariableExpr
			st varexprcast, expr_
			dim name as new Token
			st name, varexprcast.m_name
			
			dim temp as new AssignExpr
			temp.init name, value
		endif
		'raise error here

	endif
	
	st assignment_, expr_
end function

private function or_f()
	dim expr_ as new Expr
	st expr_, and_f()
	
	while (match(OR_)) 
		dim operator as new Token
		st operator, previous()
		
		dim right as new Expr
		st right, equality_()
		
		dim temp as new LogicalExpr
		dim prevExpr as new Expr
		st prevExpr, expr_
		temp.init prevExpr, operator, right
		st expr_, temp
	wend
	st or_f, expr_
end function

AssignExpr : Token name, Expr value
VariableExpr : Token name
LogicalExpr: Expr left, Token operator, Expr right 

Stmt class:

VarStmt: Token type_, Token name, Expr initializer ' optional initializer
ExprStmt:  Expr expression
PrintStmt: Expr expression
BlockStmt: Variant statements ' variant o stmts
IfStmt : Expr condition, Stmt thenBranch, Stmt elseBranch 'optional else
WhileStmt: Expr condition, Stmt body 

Parser:

public function parse() as variant 'of stmts
	dim statements as variant
	while (not isAtEnd())
		arad statements, declaration_f()
	wend
end function

private function declaration_f()
	// this is where things fail (no error handling)
	if (isTypeName(peek())) then
		st declaration_f, varDeclaration_f()
		exit function
	endif

	st declaration_f, statement_f()
end function

private function isTypeName(tk as Token)
	select case tk.m_type_
		case TYPE_:
			isTypeName = true
		case default:
			isTypeName = false
	end case 
end function

private function varDeclaration_f()
	dim varType as Token
	st varType, consume(TYPE_, "Expected type before variable name")
	dim varName as Token
	st varName, consume(IDENTIFIER_, "Expected variable name")
	dim initializer as new Expr
	
	
	dim temp as new VarStmt
	if (match(EQUAL_)) then
		st initializer, expression_()
		temp.init varType, varName, initializer	
	else
		temp.init varType, varName 'do we need a null node hmm (no thats in the codeGen, for the init function to handle the boolean)
	endif
end function

private function statement_f() 
	if (match(IF_) then
		st statement_f, ifStatement_f()
		exit function
	endif
	if (match(PRINT_)) then
		st statement_f, printStatement_f()
		exit function
	endif
	if (match(LEFT_BRACE)) then
		dim temp as new BlockStmt
		temp.init block_f()
		st statement_f, temp
		exit function
	endif 

	if(match(WHILE_)) then
		st statement_f, whileStatement_f()
		exit function
	endif
	st statement_f, expressionStatement_f()
end function

private function whileStatement_f()
	consume LEFT_PAREN, "Expected '(" after 'while'"
	dim condition as new Expr
	st condition, expression_f()
	
	consume RIGHT_PAREN, "Expected ')" after while condition"
end function

private function ifStatement_f()
	consume LEFT_PAREN, "Expected '(' after 'if'."
	dim condition as new Expr
	st condition, expresssion_f ()
	consume RIGHT_PAREN, "Expected ')' after condition"
	
	dim thenBranch as new Stmt
	dim elseBranch as Stmt
	st thenBranch, statement_f()

	dim temp as new IfStmt
	if (match(ELSE_)) then
		st elseBranch, statement_f()
		temp.init condition, thenBranch, elseBranch
	else
		temp.init condition, thenBranch
	endif
	
	st ifStatement_f, temp

end function

private function printStatement_f() 
	dim val as new Expr
	st val, expression_()
	consume SEMICOLON, "Expect ';' after value"
	dim temp as new PrintStmt
	temp.init val
	st printStatement_, temp
end function

private function expressionStatement_f()
	dim expr_ as new expr
	st expr_, expression_()
	consume SEMICOLON, "Expect ';' after value"
	dim temp as new ExprStmt
	temp.init val
	st printStatement_, temp
end function

private function block_f()
	dim statements as variant
	while (not check(RIGHT_BRACE) and not isAtEnd()) 
		arad statements, declaration_f()
	wend
	consume RIGHT_BRACE, "Expect '}' after block"
	block_f = statements
end function
